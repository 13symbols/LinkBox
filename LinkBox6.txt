        // Жёсткая очистка URL от запятых, точек, скобок и мусора на конце (исправление бага ",,")
        function cleanRawUrl(raw) {
            let u = (raw || '').trim();
            // Срезаем любые запятые, точки, кавычки, скобки и точки с запятой с концов ссылки
            u = u.replace(/^[,\s"'<>()[\]{};:]+|[,\s"'<>()[\]{}.;:]+$/g, '');
            if (!u) return '';
            if (!/^https?:\/\//i.test(u)) u = 'https://' + u;
            return u;
        }

        function saveStateToStorage() {
            try {
                const payload = {
                    items: ITEMS,
                    lockedFolders: Array.from(LOCKED_FOLDERS),
                    themeMode,
                    viewMode,
                    currentLang
                };
                localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));
            } catch (e) {}
        }

        function loadStateFromStorage() {
            try {
                const raw = localStorage.getItem(STORAGE_KEY) || localStorage.getItem('linkbox_hyperos_v1');
                if (!raw) return false;
                const parsed = JSON.parse(raw);
                if (Array.isArray(parsed.items)) {
                    // Заодно лечим старые ссылки с запятыми на конце, если они были сохранены
                    ITEMS = parsed.items.map(it => ({
                        ...it,
                        url: cleanRawUrl(it.url)
                    }));
                }
                if (Array.isArray(parsed.lockedFolders)) LOCKED_FOLDERS = new Set(parsed.lockedFolders);
                if (parsed.themeMode) themeMode = parsed.themeMode;
                if (parsed.viewMode) viewMode = parsed.viewMode;
                if (parsed.currentLang) currentLang = parsed.currentLang;
                return true;
            } catch (e) {
                return false;
            }
        }

        function exportBrowserBookmarks() {
            const foldersMap = {};
            ITEMS.forEach(item => {

                foldersMap[item.cat].push(item);
            });

                html += `    <DT><H3 ADD_DATE="${nowUnix}">${cat}</H3>\n    <DL><p>\n`;

        // Нормализация URL для точного поиска дубликатов (с предварительной очисткой запятых)
        function normalizeUrlForCheck(rawUrl) {
            const u = cleanRawUrl(rawUrl);
            if (!u) return '';
            try {
                const parsed = new URL(u);
                const host = parsed.hostname.replace(/^www\./i, '').toLowerCase();
                const path = parsed.pathname.replace(/\/+$/, '');
                return host + path + parsed.search.toLowerCase();
            } catch (e) {
                return u.replace(/^https?:\/\//i, '').replace(/^www\./i, '').replace(/\/+$/, '').toLowerCase();
            }
        }

        function findDuplicateItem(rawUrl) {
            const norm = normalizeUrlForCheck(rawUrl);
            if (!norm) return null;
            return ITEMS.find(item => normalizeUrlForCheck(item.url) === norm) || null;
        }

        // Извлечение красивого имени бренда из домена (без поддомена "m." или "ukr.")
        function extractCleanBrandName(domain) {
            const parts = domain.split('.').filter(p => !['www', 'm', 'mobile', 'ua', 'com', 'net', 'org', 'gov', 'edu', 'info', 'io', 'co', 'uk', 'ru', 'biz'].includes(p));
            const root = parts.length > 0 ? parts[parts.length - 1] : domain.split('.')[0];
            if (!root) return 'Сайт';
            return root.charAt(0).toUpperCase() + root.slice(1);
        }

        // Очистка человеческого заголовка (если из браузера пришло нормальное название, оставляем его; если голый URL или мусор — не берём)
        function sanitizeBookmarkTitle(rawTitle, cleanName) {
            if (!rawTitle) return '';
            let tStr = rawTitle.trim();
            if (/^https?:\/\//i.test(tStr)) return '';
            // Отсекаем случайные хвосты из запятых
            tStr = tStr.replace(/[,\s]+$/g, '');
            if (tStr.length < 2) return '';
            return tStr;
        }

        // Умный ИИ-классификатор: раскладывает сайты по 11 осмысленным категориям и пишет живую контекстную заметку
        function buildSmartAIItem(rawUrl, customTitle = '') {
            const finalUrl = cleanRawUrl(rawUrl);
            const domain = finalUrl.replace(/^https?:\/\//i, '').replace(/^www\./i, '').split('/')[0].toLowerCase();
            const cleanName = extractCleanBrandName(domain);
            const humanTitle = sanitizeBookmarkTitle(customTitle, cleanName);
            const combinedSignal = `${finalUrl} ${humanTitle}`.toLowerCase();

            let cat = currentLang === 'uk' ? 'Веб-ресурси' : 'Web Resources';
            let color = '#4F6352';
            let svg = LINE_ICONS.globe;
            let tags = [`#${cleanName.toLowerCase()}`, '#веб'];
            let defaultCategoryTitle = currentLang === 'uk' ? `${cleanName} — Корисний ресурс` : `${cleanName} — Web Resource`;
            let note = '';

            // 1. МУЗЫКА И АУДИО
            if (/spotify|soundcloud|music\.apple|deezer|bandcamp|last\.fm|shazam|mp3|radio|podcast|audio|музик|пісн|альбом|плейлист/.test(combinedSignal)) {
                cat = currentLang === 'uk' ? 'Музика' : 'Music';
                color = '#3F6A4E';
                svg = LINE_ICONS.music;
                tags = ['#музика', '#аудіо', `#${cleanName.toLowerCase()}`];
                defaultCategoryTitle = currentLang === 'uk' ? `${cleanName} — Музика та аудіо` : `${cleanName} — Music & Audio`;
                note = humanTitle
                    ? `Музична добірка або трек «${humanTitle}» на ${domain} для прослуховування.`
                    : `Аудіострімінг та музичний каталог на платформі ${cleanName} (${domain}).`;
            }
            // 2. КИНО, СЕРИАЛЫ И ВИДЕО
            else if (/youtube|youtu\.be|netflix|megogo|sweet\.tv|takflix|kinopoisk|imdb|vimeo|twitch|rezka|filmix|uaserial|serial|кіно|серіал|фільм|мульт|movie|cinema|video/.test(combinedSignal)) {
                cat = currentLang === 'uk' ? 'Кіно та Серіали' : 'Movies & Video';
                color = '#7D3F3B';
                svg = LINE_ICONS.video;
                tags = ['#кіно', '#серіали', `#${cleanName.toLowerCase()}`];
                defaultCategoryTitle = currentLang === 'uk' ? `${cleanName} — Кіно та відео` : `${cleanName} — Video & Series`;
                note = humanTitle
                    ? `Відео або серіал «${humanTitle}» (${domain}) — збережено у список на перегляд.`
                    : `Відеоплатформа ${cleanName}: кіно, серіали та ефіри для вечірнього перегляду.`;

        // Разбор одной или сразу пачки ссылок через главное поле ИИ
        function addLinkByAI() {
            const input = document.getElementById('url-input');
            const rawText = input.value.trim();
            if (!rawText) return;

            const urlMatches = rawText.match(/(?:https?:\/\/|www\.)[^\s"'<>]+/gi);
            const rawCandidates = urlMatches && urlMatches.length > 0 ? urlMatches : [rawText];
            const candidates = rawCandidates.map(u => cleanRawUrl(u)).filter(Boolean);

            if (candidates.length === 1) {
                const singleUrl = candidates[0];
                const existing = findDuplicateItem(singleUrl);
                if (existing) {
                    showToast(`${t('duplicateFound')}${existing.name} (${existing.cat})`);
                    highlightExistingCard(existing);
                    return;
                }
                const newItem = buildSmartAIItem(singleUrl);
                ITEMS.unshift(newItem);
                input.value = '';
                saveStateToStorage();
                renderApp();
                showToast(t('aiAddedSingle').replace('{cat}', newItem.cat));
            } else if (candidates.length > 1) {
                let added = 0;
                let dups = 0;
                candidates.forEach(u => {
                    if (findDuplicateItem(u)) {
                        dups++;
                    } else {
                        ITEMS.unshift(buildSmartAIItem(u));
                        added++;
                    }
                });
                input.value = '';
                saveStateToStorage();
                renderApp();
                showToast(t('aiImportReport').replace('{added}', added).replace('{dups}', dups));
            }
        }

        function handleAIFileImport(event) {
            const file = event.target.files?.[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = (e) => {
                const content = String(e.target?.result || '');
                const extracted = [];

                const parser = new DOMParser();
                const doc = parser.parseFromString(content, 'text/html');
                const anchors = doc.querySelectorAll('a[href]');

                if (anchors.length > 0) {
                    anchors.forEach(a => {
                        const href = cleanRawUrl(a.getAttribute('href') || '');
                        if (/^https?:\/\//i.test(href)) {
                            extracted.push({ url: href, title: (a.textContent || '').trim() });
                        }
                    });
                } else {
                    const rawUrls = content.match(/https?:\/\/[^\s"'<>]+/gi) || [];
                    rawUrls.forEach(u => {
                        const clean = cleanRawUrl(u);
                        if (clean) extracted.push({ url: clean, title: '' });
                    });
                }

                let added = 0;
                let dups = 0;

                extracted.forEach(entry => {
                    if (findDuplicateItem(entry.url)) {
                        dups++;
                    } else {
                        ITEMS.unshift(buildSmartAIItem(entry.url, entry.title));
                        added++;
                    }
                });

                closeMenuSheet();
                saveStateToStorage();
                renderApp();
                showToast(t('aiImportReport').replace('{added}', added).replace('{dups}', dups));
            };
            reader.readAsText(file);
        }

        function attachSheetDrag(handleId, panelId, closeFn) {
            const handle = document.getElementById(handleId);
            const panel = document.getElementById(panelId);
            let startY = 0, currentY = 0, dragging = false;

            function onStart(e) {
                dragging = true;
                startY = e.type.includes('mouse') ? e.clientY : e.touches[0].clientY;
                panel.style.transition = 'none';
            }
            function onMove(e) {
                if (!dragging) return;
                currentY = e.type.includes('mouse') ? e.clientY : e.touches[0].clientY;
                const diff = currentY - startY;
                if (diff > 0) panel.style.transform = `translateY(${diff}px)`;
            }
            function onEnd() {
                if (!dragging) return;
                dragging = false;
                panel.style.transition = 'transform 0.34s cubic-bezier(0.16, 1, 0.3, 1)';
                if (currentY - startY > 90) closeFn();
                else panel.style.transform = 'translateY(0)';
                startY = 0; currentY = 0;
            }

            handle.addEventListener('mousedown', onStart);
            window.addEventListener('mousemove', onMove);
            window.addEventListener('mouseup', onEnd);

            handle.addEventListener('touchstart', onStart, { passive: true });
            window.addEventListener('touchmove', onMove, { passive: true });
            window.addEventListener('touchend', onEnd);
        }
